					has_country_flag = PRC_gov_finds_out_BD
				}
			}

			has_country_flag = PRC_NS_boosted_gov
				80 = {
					limit = { }
					add_to_variable = {
						var = PRC_gov_sus
						value = 1
					}
			 }
				20 = {
					limit = { }
					add_to_variable = {
						var = PRC_gov_sus
						value = 1
					}
					add_to_variable = {
						var = PRC_NS_gang_power
						value = -1
					}
			 }			 
			}			
		else = {
		random_list = {
				50 = {
					limit = { }
					add_to_variable = {
						var = PRC_gov_sus
						value = 1
					}
			 }
				50 = {
					limit = { }
					add_to_variable = {
						var = PRC_gov_sus
						value = 1
					}
					add_to_variable = {
						var = PRC_NS_gang_power
						value = -1
					}
			 }			 
			}
		}	
}

			if = {
				limit = {
					check_variable = {
						var = PRC_gov_sus
						value = 10

					}
				}

			}
}
